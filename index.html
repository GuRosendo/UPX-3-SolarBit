    // ESP32 + sensor SCT-013 (corrente), envia a cada 2 s
#include <WiFi.h>
#include <HTTPClient.h>
void loop() {
  float amps = lerCorrente();      // leitura do SCT-013
  float watts = amps * 127.0;      // tensão da rede
  HTTPClient http;
  http.begin("http://SEU_SERVIDOR/api/leitura");
  http.addHeader("Content-Type", "application/json");
  http.POST("{\"watts\":" + String(watts) + "}");
  http.end();
  delay(2000);
}

    Nesta demonstração os valores são simulados a partir dos aparelhos cadastrados. Com o hardware real, o ESP32 envia as leituras para o mesmo painel.
